<?php

declare(strict_types=1);

namespace App\Core;

use App\Core\Exceptions\ExceptionHandler;
use App\Core\Middleware\MiddlewareInterface;
use Throwable;

class Application
{
    protected Router $router;
    /** @var MiddlewareInterface[] */
    protected array $middleware = [];
    protected string $basePath;

    public function __construct(string $basePath)
    {
        $this->basePath = rtrim($basePath, '/\\');
        $this->router = new Router();
    }

    public function getRouter(): Router
    {
        return $this->router;
    }

    public function addMiddleware(MiddlewareInterface $middleware): self
    {
        $this->middleware[] = $middleware;
        return $this;
    }

    public function getBasePath(): string
    {
        return $this->basePath;
    }

    /**
     * Dispatch an incoming Request through middleware pipeline and Router.
     */
    public function dispatch(Request $request): Response
    {
        try {
            return $this->runPipeline($request);
        } catch (Throwable $e) {
            return ExceptionHandler::handle($e, $request);
        }
    }

    /**
     * Execute the middleware pipeline and resolve the route handler.
     */
    protected function runPipeline(Request $request): Response
    {
        $middlewareStack = $this->middleware;

        $corePipeline = function (Request $req): Response {
            $match = $this->router->match($req);
            $handler = $match['handler'];
            $params = $match['params'];

            return $this->invokeHandler($handler, $req, $params);
        };

        $pipeline = array_reduce(
            array_reverse($middlewareStack),
            function (callable $next, MiddlewareInterface $middleware) {
                return function (Request $req) use ($middleware, $next): Response {
                    return $middleware->handle($req, $next);
                };
            },
            $corePipeline
        );

        return $pipeline($request);
    }

    /**
     * Invoke the matched route handler.
     *
     * @param callable|array{0: class-string, 1: string} $handler
     * @param array<string, string> $params
     */
    protected function invokeHandler(callable|array $handler, Request $request, array $params): Response
    {
        if (is_array($handler) && is_string($handler[0])) {
            $class = $handler[0];
            $method = $handler[1];
            $instance = new $class();
            $result = $instance->$method($request, ...array_values($params));
        } else {
            $result = call_user_func($handler, $request, ...array_values($params));
        }

        if ($result instanceof Response) {
            return $result;
        }

        if (is_array($result)) {
            return Response::json($result);
        }

        if (is_string($result)) {
            return new Response($result);
        }

        return Response::json($result);
    }
}

declare(strict_types=1);

namespace App\Core;

use App\Core\Exceptions\BadRequestException;

class Request
{
    protected string $method;
    protected string $path;
    /** @var array<string, mixed> */
    protected array $queryParams;
    /** @var array<string, mixed> */
    protected array $bodyParams;
    /** @var array<string, string> */
    protected array $routeParams = [];
    /** @var array<string, string> */
    protected array $headers;
    protected string $rawBody;

    /**
     * @param array<string, mixed> $queryParams
     * @param array<string, mixed> $bodyParams
     * @param array<string, string> $headers
     */
    public function __construct(
        string $method = 'GET',
        string $path = '/',
        array $queryParams = [],
        array $bodyParams = [],
        array $headers = [],
        string $rawBody = ''
    ) {
        $this->method = strtoupper(trim($method));
        $this->path = self::normalizePath($path);
        $this->queryParams = $queryParams;
        $this->bodyParams = $bodyParams;
        $this->headers = self::normalizeHeaders($headers);
        $this->rawBody = $rawBody;
    }

    /**
     * Parse raw JSON payload into an array of parameters.
     *
     * @return array<string, mixed>
     * @throws BadRequestException
     */
    public static function parseJsonPayload(string $rawBody): array
    {
        if (trim($rawBody) === '') {
            return [];
        }

        $decoded = json_decode($rawBody, true);
        if (json_last_error() !== JSON_ERROR_NONE) {
            throw new BadRequestException('Malformed JSON in request payload: ' . json_last_error_msg());
        }

        if (!is_array($decoded)) {
            throw new BadRequestException('Malformed JSON in request payload: Payload must be a JSON object or array');
        }

        return $decoded;
    }

    /**
     * Capture the current HTTP request from superglobals and input stream.
     */
    public static function capture(?string $rawBody = null): self
    {
        $server = $_SERVER;
        $method = $server['REQUEST_METHOD'] ?? 'GET';
        $uri = $server['REQUEST_URI'] ?? '/';

        $parsedUrl = parse_url($uri, PHP_URL_PATH);
        $path = is_string($parsedUrl) ? $parsedUrl : '/';

        $headers = [];
        if (function_exists('getallheaders')) {
            $rawHeaders = getallheaders();
            if (is_array($rawHeaders)) {
                $headers = $rawHeaders;
            }
        } else {
            foreach ($server as $key => $value) {
                if (str_starts_with($key, 'HTTP_')) {
                    $headerName = str_replace('_', '-', substr($key, 5));
                    $headers[$headerName] = (string) $value;
                } elseif (in_array($key, ['CONTENT_TYPE', 'CONTENT_LENGTH', 'CONTENT_MD5'], true)) {
                    $headerName = str_replace('_', '-', $key);
                    $headers[$headerName] = (string) $value;
                }
            }
        }

        if ($rawBody === null) {
            $input = file_get_contents('php://input');
            $rawBody = $input !== false ? $input : '';
        }

        // Case-insensitive Content-Type detection across server vars and headers
        $contentType = $server['CONTENT_TYPE'] ?? $server['HTTP_CONTENT_TYPE'] ?? '';
        if ($contentType === '') {
            foreach ($headers as $k => $v) {
                if (strcasecmp($k, 'Content-Type') === 0) {
                    $contentType = (string) $v;
                    break;
                }
            }
        }

        $bodyParams = [];

        if (str_contains(strtolower($contentType), 'application/json')) {
            $bodyParams = self::parseJsonPayload($rawBody);
        } elseif (!empty($_POST)) {
            $bodyParams = $_POST;
        }

        return new self($method, $path, $_GET, $bodyParams, $headers, $rawBody);
    }

    public static function normalizePath(string $path): string
    {
        $trimmed = '/' . trim($path, '/');
        return $trimmed === '' ? '/' : $trimmed;
    }

    public function getMethod(): string
    {
        return $this->method;
    }

    public function getPath(): string
    {
        return $this->path;
    }

    /**
     * @return array<string, mixed>
     */
    public function getQueryParams(): array
    {
        return $this->queryParams;
    }

    public function query(string $key, mixed $default = null): mixed
    {
        return $this->queryParams[$key] ?? $default;
    }

    /**
     * @return array<string, mixed>
     */
    public function getBodyParams(): array
    {
        return $this->bodyParams;
    }

    public function input(string $key, mixed $default = null): mixed
    {
        return $this->bodyParams[$key] ?? $this->queryParams[$key] ?? $default;
    }

    /**
     * @return array<string, mixed>
     */
    public function all(): array
    {
        return array_merge($this->queryParams, $this->bodyParams);
    }

    public function getRawBody(): string
    {
        return $this->rawBody;
    }

    /**
     * @return array<string, string>
     */
    public function getHeaders(): array
    {
        return $this->headers;
    }

    public function getHeader(string $name, ?string $default = null): ?string
    {
        $lower = strtolower($name);
        return $this->headers[$lower] ?? $default;
    }

    /**
     * @param array<string, string> $params
     */
    public function setRouteParams(array $params): void
    {
        $this->routeParams = $params;
    }

    /**
     * @return array<string, string>
     */
    public function getRouteParams(): array
    {
        return $this->routeParams;
    }

    public function param(string $key, ?string $default = null): ?string
    {
        return $this->routeParams[$key] ?? $default;
    }

    /**
     * @param array<string, string> $headers
     * @return array<string, string>
     */
    private static function normalizeHeaders(array $headers): array
    {
        $normalized = [];
        foreach ($headers as $key => $value) {
            $normalized[strtolower((string) $key)] = (string) $value;
        }
        return $normalized;
    }
}

<?php

declare(strict_types=1);

namespace App\Core;

class Response
{
    protected int $statusCode;
    /** @var array<string, string> */
    protected array $headers;
    protected string $content;

    /**
     * @param array<string, string> $headers
     */
    public function __construct(string $content = '', int $statusCode = 200, array $headers = [])
    {
        $this->content = $content;
        $this->statusCode = $statusCode;
        $this->headers = array_merge([
            'Content-Type' => 'application/json; charset=utf-8',
            'X-Content-Type-Options' => 'nosniff',
            'X-Frame-Options' => 'DENY',
            'X-XSS-Protection' => '1; mode=block',
            'Referrer-Policy' => 'strict-origin-when-cross-origin',
        ], $headers);
    }

    /**
     * Create a standard JSON response.
     *
     * @param array<string, string> $headers
     */
    public static function json(mixed $data, int $statusCode = 200, array $headers = []): self
    {
        $encoded = json_encode($data, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES);
        if ($encoded === false) {
            $encoded = json_encode([
                'success' => false,
                'error' => [
                    'code' => 'JSON_ENCODE_ERROR',
                    'message' => 'Failed to encode JSON response',
                ],
            ]);
            $statusCode = 500;
        }

        return new self($encoded, $statusCode, $headers);
    }

    /**
     * Create a structured error response.
     *
     * @param array<string, mixed> $errors
     * @param array<string, string> $headers
     */
    public static function error(
        string $message,
        int $statusCode = 400,
        array $errors = [],
        ?string $code = null,
        array $headers = []
    ): self {
        $body = [
            'success' => false,
            'error' => [
                'code' => $code ?? self::getDefaultErrorCode($statusCode),
                'message' => $message,
            ],
        ];

        if (!empty($errors)) {
            $body['error']['details'] = $errors;
        }

        return self::json($body, $statusCode, $headers);
    }

    public function getStatusCode(): int
    {
        return $this->statusCode;
    }

    public function getContent(): string
    {
        return $this->content;
    }

    /**
     * @return array<string, string>
     */
    public function getHeaders(): array
    {
        return $this->headers;
    }

    public function withHeader(string $name, string $value): self
    {
        $this->headers[$name] = $value;
        return $this;
    }

    /**
     * Send headers and body to HTTP client.
     */
    public function send(): void
    {
        if (!headers_sent()) {
            http_response_code($this->statusCode);
            foreach ($this->headers as $name => $value) {
                header("{$name}: {$value}");
            }
        }

        echo $this->content;
    }

    private static function getDefaultErrorCode(int $statusCode): string
    {
        return match ($statusCode) {
            400 => 'BAD_REQUEST',
            401 => 'UNAUTHORIZED',
            403 => 'FORBIDDEN',
            404 => 'NOT_FOUND',
            405 => 'METHOD_NOT_ALLOWED',
            409 => 'CONFLICT',
            422 => 'UNPROCESSABLE_ENTITY',
            429 => 'TOO_MANY_REQUESTS',
            default => 'INTERNAL_SERVER_ERROR',
        };
    }
}

<?php

declare(strict_types=1);

namespace App\Core;

use App\Core\Exceptions\MethodNotAllowedException;
use App\Core\Exceptions\NotFoundException;

class Router
{
    /**
     * @var array<int, array{
     *     method: string,
     *     path: string,
     *     regex: string,
     *     paramNames: string[],
     *     handler: callable|array{0: class-string, 1: string}
     * }>
     */
    protected array $routes = [];

    public function addRoute(string $method, string $path, callable|array $handler): self
    {
        $normalizedPath = Request::normalizePath($path);
        [$regex, $paramNames] = self::compileRoute($normalizedPath);

        $this->routes[] = [
            'method' => strtoupper($method),
            'path' => $normalizedPath,
            'regex' => $regex,
            'paramNames' => $paramNames,
            'handler' => $handler,
        ];

        return $this;
    }

    public function get(string $path, callable|array $handler): self
    {
        return $this->addRoute('GET', $path, $handler);
    }

    public function post(string $path, callable|array $handler): self
    {
        return $this->addRoute('POST', $path, $handler);
    }

    public function put(string $path, callable|array $handler): self
    {
        return $this->addRoute('PUT', $path, $handler);
    }

    public function patch(string $path, callable|array $handler): self
    {
        return $this->addRoute('PATCH', $path, $handler);
    }

    public function delete(string $path, callable|array $handler): self
    {
        return $this->addRoute('DELETE', $path, $handler);
    }

    public function options(string $path, callable|array $handler): self
    {
        return $this->addRoute('OPTIONS', $path, $handler);
    }

    /**
     * Match a request against registered routes.
     *
     * @throws NotFoundException
     * @throws MethodNotAllowedException
     * @return array{handler: callable|array, params: array<string, string>}
     */
    public function match(Request $request): array
    {
        $requestMethod = $request->getMethod();
        $requestPath = $request->getPath();

        $matchingPathFound = false;
        $allowedMethods = [];

        foreach ($this->routes as $route) {
            if (preg_match($route['regex'], $requestPath, $matches)) {
                $matchingPathFound = true;
                $allowedMethods[] = $route['method'];

                if ($route['method'] === $requestMethod) {
                    $params = [];
                    foreach ($route['paramNames'] as $paramName) {
                        if (isset($matches[$paramName])) {
                            $params[$paramName] = $matches[$paramName];
                        }
                    }

                    $request->setRouteParams($params);

                    return [
                        'handler' => $route['handler'],
                        'params' => $params,
                    ];
                }
            }
        }

        if ($matchingPathFound) {
            $uniqueMethods = array_values(array_unique($allowedMethods));
            throw new MethodNotAllowedException(
                $uniqueMethods,
                "HTTP method {$requestMethod} is not supported for path {$requestPath}"
            );
        }

        throw new NotFoundException("Route not found: {$requestMethod} {$requestPath}");
    }

    /**
     * Compile a path with parameters (e.g. /games/{id}) into a regex.
     *
     * @return array{0: string, 1: string[]}
     */
    protected static function compileRoute(string $path): array
    {
        $paramNames = [];
        $pattern = preg_replace_callback('/\{([a-zA-Z0-9_]+)\}/', function ($matches) use (&$paramNames) {
            $paramNames[] = $matches[1];
            return '(?P<' . $matches[1] . '>[^/]+)';
        }, $path);

        $regex = '#^' . $pattern . '$#u';

        return [$regex, $paramNames];
    }
}

<?php

declare(strict_types=1);

namespace App\Core\Exceptions;

use App\Core\Request;
use App\Core\Response;
use App\Infrastructure\Config\Config;
use Throwable;

class ExceptionHandler
{
    /** @var (callable(string): void)|null */
    protected static $logger = null;

    /**
     * Set a custom logger callback (useful for testing or custom log channels).
     *
     * @param (callable(string): void)|null $logger
     */
    public static function setLogger(?callable $logger): void
    {
        self::$logger = $logger;
    }

    /**
     * Convert an uncaught Throwable into a clean JSON Response.
     */
    public static function handle(Throwable $e, ?Request $request = null): Response
    {
        $statusCode = 500;
        $headers = [];
        $errors = [];
        $errorCode = 'INTERNAL_SERVER_ERROR';

        if ($e instanceof HttpException) {
            $statusCode = $e->getStatusCode();
            $headers = $e->getHeaders();
            $message = $e->getMessage();

            if ($e instanceof BadRequestException) {
                $errorCode = 'BAD_REQUEST';
                $errors = $e->getErrors();
            } elseif ($e instanceof NotFoundException) {
                $errorCode = 'NOT_FOUND';
            } elseif ($e instanceof MethodNotAllowedException) {
                $errorCode = 'METHOD_NOT_ALLOWED';
            }
        } else {
            $isDebug = (bool) Config::get('APP_DEBUG', false);
            if ($isDebug) {
                $message = self::sanitize($e->getMessage());
                $errors = [
                    'exception' => get_class($e),
                    'file' => $e->getFile(),
                    'line' => $e->getLine(),
                ];
            } else {
                $message = 'Internal Server Error';
            }

            self::logError($e);
        }

        return Response::error($message, $statusCode, $errors, $errorCode, $headers);
    }

    /**
     * Format a safe log message with structured metadata and sanitized secrets across the entire exception chain.
     */
    public static function formatLogMessage(Throwable $e): string
    {
        $frames = [];
        $current = $e;

        while ($current !== null) {
            $safeMessage = self::sanitize($current->getMessage());
            $safeTrace = self::sanitize($current->getTraceAsString());

            $frames[] = sprintf(
                "[%s] %s: %s in %s:%d\nTrace:\n%s",
                date('Y-m-d H:i:s'),
                get_class($current),
                $safeMessage,
                $current->getFile(),
                $current->getLine(),
                $safeTrace
            );

            $current = $current->getPrevious();
        }

        return implode("\nCaused by: ", $frames) . "\n";
    }

    /**
     * Sanitize sensitive keywords and patterns from string output.
     */
    public static function sanitize(string $input): string
    {
        // 1. Redact URI/DSN passwords: scheme://user:password@host
        $sanitized = preg_replace(
            '#://([^:]+):([^@]+)@#',
            '://$1:***@',
            $input
        );
        if (!is_string($sanitized)) {
            $sanitized = $input;
        }

        // 2. Redact Authorization headers: Authorization: Bearer <token> or Basic <token>
        $sanitized = preg_replace(
            '/((?:authorization|proxy-authorization)\s*:\s*(?:bearer|basic|token)\s+)[^\s,;\r\n]+/i',
            '$1***',
            $sanitized
        );
        if (!is_string($sanitized)) {
            $sanitized = $input;
        }

        // 3. Redact URL query parameters: ?token=xyz or &password=xyz or &key=xyz
        $sanitized = preg_replace(
            '/([?&](?:password|pwd|token|client_secret|secret|api_?key|key|access_token|refresh_token|auth)=)[^&\s#]+/i',
            '$1***',
            $sanitized
        );
        if (!is_string($sanitized)) {
            $sanitized = $input;
        }

        // 4. Redact key-value pairs (JSON quoted, array dumps, key=val, key: val, or key val)
        // Matches:
        // - Prefix: optional [ or [' or [" then keyword then optional ] or '] or "]
        // - Separator: => or : or is or = or whitespace
        // - Value: single-quoted string, double-quoted string, or unquoted value up to delimiter/whitespace
        $pattern = '/(?P<prefix>\[?(?:\'|")?\b(?:password|pwd|token|client_secret|secret|api_?key|access_token|refresh_token|auth|credential)\b(?:\'|")?\]?)(?P<sep>\s*(?:=>|:|is|=|\s+)\s*)(?P<val>\'[^\']*\'|"[^"]*"|(?!(?:password|pwd|token|client_secret|secret|api_?key|access_token|refresh_token|auth|credential)\s*(?:=|:|=>))[^\s,\r\n]+)/i';
        $sanitized = preg_replace_callback($pattern, function (array $m): string {
            $val = $m['val'];
            // If already redacted, avoid altering
            if ($val === '***' || $val === '"***"' || $val === "'***'" || str_starts_with($val, '***')) {
                return $m[0];
            }

            if (str_starts_with($val, '"') && str_ends_with($val, '"')) {
                $replacementVal = '"***"';
            } elseif (str_starts_with($val, "'") && str_ends_with($val, "'")) {
                $replacementVal = "'***'";
            } else {
                $replacementVal = '***';
            }

            return $m['prefix'] . $m['sep'] . $replacementVal;
        }, $sanitized);

        if (!is_string($sanitized)) {
            $sanitized = $input;
        }

        return $sanitized;
    }

    /**
     * Log unexpected server errors safely.
     */
    protected static function logError(Throwable $e): void
    {
        $logMessage = self::formatLogMessage($e);

        if (self::$logger !== null) {
            (self::$logger)($logMessage);
        } else {
            error_log($logMessage);
        }
    }
}

<?php

declare(strict_types=1);

namespace App\Core\Middleware;

use App\Core\Request;
use App\Core\Response;
use App\Infrastructure\Config\Config;

class CorsMiddleware implements MiddlewareInterface
{
    /** @var string[] */
    protected array $allowedOrigins;
    protected bool $supportsCredentials;

    /**
     * @param string[]|null $allowedOrigins
     * @param bool|null $supportsCredentials
     */
    public function __construct(?array $allowedOrigins = null, ?bool $supportsCredentials = null)
    {
        if ($allowedOrigins !== null) {
            $this->allowedOrigins = array_values(array_filter(array_map('trim', $allowedOrigins)));
        } else {
            $rawOrigins = Config::get('ALLOWED_ORIGINS', 'http://127.0.0.1:3000,http://localhost:3000');
            $this->allowedOrigins = array_values(array_filter(array_map('trim', explode(',', (string) $rawOrigins))));
        }

        if ($supportsCredentials !== null) {
            $this->supportsCredentials = $supportsCredentials;
        } else {
            $this->supportsCredentials = (bool) Config::get('CORS_SUPPORTS_CREDENTIALS', true);
        }
    }

    public function handle(Request $request, callable $next): Response
    {
        $origin = $request->getHeader('origin');
        $isWildcard = in_array('*', $this->allowedOrigins, true);

        // Security requirement: If wildcard is present, credentials MUST NOT be enabled
        $allowCredentials = $this->supportsCredentials && !$isWildcard;

        $isAllowed = false;
        $matchedOrigin = '';

        if ($origin !== null && $origin !== '') {
            if ($isWildcard) {
                // Literal wildcard: never reflect dynamic origin with credentials
                $isAllowed = true;
                $matchedOrigin = '*';
            } else {
                foreach ($this->allowedOrigins as $allowed) {
                    if (strcasecmp($allowed, $origin) === 0) {
                        $isAllowed = true;
                        $matchedOrigin = $origin;
                        break;
                    }
                }
            }
        }

        // Handle preflight OPTIONS request
        if ($request->getMethod() === 'OPTIONS') {
            $headers = [
                'Access-Control-Allow-Methods' => 'GET, POST, PUT, PATCH, DELETE, OPTIONS',
                'Access-Control-Allow-Headers' => 'Content-Type, Authorization, X-Requested-With, Accept, Idempotency-Key',
                'Access-Control-Max-Age' => '86400',
            ];

            if ($isAllowed) {
                $headers['Access-Control-Allow-Origin'] = $matchedOrigin;
                if ($allowCredentials) {
                    $headers['Access-Control-Allow-Credentials'] = 'true';
                }
            }

            $response = new Response('', 204, $headers);
            // RFC 7234: If the response representation varies by Origin allowlist,
            // Vary: Origin must be set consistently on all responses to protect HTTP caches.
            if (!$isWildcard) {
                $response = $this->appendVaryHeader($response, 'Origin');
            }

            return $response;
        }

        /** @var Response $response */
        $response = $next($request);

        // For non-wildcard configurations, response headers depend on the Origin header.
        // Therefore, Vary: Origin MUST be set on all responses (including requests without Origin,
        // or with disallowed origins) so intermediate caches do not serve an origin-specific response
        // to an origin-less or different-origin request.
        if (!$isWildcard) {
            $response = $this->appendVaryHeader($response, 'Origin');
        }

        if ($isAllowed) {
            $response->withHeader('Access-Control-Allow-Origin', $matchedOrigin);
            if ($allowCredentials) {
                $response->withHeader('Access-Control-Allow-Credentials', 'true');
            }
        }

        return $response;
    }

    /**
     * Safely append a field to the Vary header, preserving any existing values.
     */
    protected function appendVaryHeader(Response $response, string $field): Response
    {
        $headers = $response->getHeaders();
        $existingVary = null;

        foreach ($headers as $name => $value) {
            if (strcasecmp($name, 'Vary') === 0) {
                $existingVary = $value;
                break;
            }
        }

        if ($existingVary === null || trim($existingVary) === '') {
            return $response->withHeader('Vary', $field);
        }

        $parts = array_map('trim', explode(',', $existingVary));
        foreach ($parts as $part) {
            if (strcasecmp($part, $field) === 0) {
                return $response;
            }
        }

        return $response->withHeader('Vary', $existingVary . ', ' . $field);
    }

namespace App\Features\Auth;

use App\Core\Exceptions\HttpException;

class AuthService
{
    private AuthRepository $repository;

    public function __construct(?AuthRepository $repository = null)
    {
        $this->repository = $repository ?? new AuthRepository();
    }

    /**
     * @param array<string, mixed> $input
     * @return array<string, mixed>
     */
    public function register(array $input): array
    {
        $fullName = trim((string) ($input['full_name'] ?? ''));
        $email = strtolower(trim((string) ($input['email'] ?? '')));
        $password = (string) ($input['password'] ?? '');
        $phoneRaw = $input['phone'] ?? null;
        $phone = is_string($phoneRaw) && trim($phoneRaw) !== '' ? trim($phoneRaw) : null;

        if ($fullName === '' || mb_strlen($fullName) < 2 || mb_strlen($fullName) > 120) {
            throw new HttpException('Họ tên phải từ 2 đến 120 ký tự', 400);
        }

        if ($email === '' || filter_var($email, FILTER_VALIDATE_EMAIL) === false || strlen($email) > 160) {
            throw new HttpException('Email không hợp lệ', 400);
        }

        if (strlen($password) < 8 || strlen($password) > 72) {
            throw new HttpException('Mật khẩu phải từ 8 đến 72 ký tự', 400);
        }

        if ($phone !== null && preg_match('/^[0-9+\-\s]{9,15}$/', $phone) !== 1) {
            throw new HttpException('Số điện thoại không hợp lệ', 400);
        }

        if ($this->repository->findByEmail($email) !== null) {
            throw new HttpException('Email đã được đăng ký', 409);
        }

        $passwordHash = password_hash($password, PASSWORD_BCRYPT);
        $userId = $this->repository->create($fullName, $email, $passwordHash, $phone);
        $user = $this->repository->findById($userId);

        if ($user === null) {
            throw new HttpException('Không thể khởi tạo tài khoản', 500);
        }

        $this->startSessionForUser($user);

        return $this->formatUser($user);
    }

    /**
     * @param array<string, mixed> $input
     * @return array<string, mixed>
     */
    public function login(array $input): array
    {
        $email = strtolower(trim((string) ($input['email'] ?? '')));
        $password = (string) ($input['password'] ?? '');

        if ($email === '' || $password === '') {
            throw new HttpException('Vui lòng nhập đầy đủ email và mật khẩu', 400);
        }

        $user = $this->repository->findByEmail($email);

        // Trả thông báo chung để tránh lộ email nào có tồn tại trong hệ thống
        if ($user === null || !password_verify($password, (string) $user['password_hash'])) {
            throw new HttpException('Email hoặc mật khẩu không chính xác', 401);
        }

        if ((int) $user['is_active'] !== 1) {
            throw new HttpException('Tài khoản của bạn đã bị khóa', 403);
        }

        $this->startSessionForUser($user);

        return $this->formatUser($user);
    }

    public function logout(): void
    {
        $_SESSION = [];

        if (session_status() === PHP_SESSION_ACTIVE) {
            if (ini_get('session.use_cookies') && !headers_sent()) {
                $params = session_get_cookie_params();
                setcookie(
                    session_name(),
                    '',
                    [
                        'expires' => time() - 42000,
                        'path' => $params['path'],
                        'domain' => $params['domain'],
                        'secure' => (bool) $params['secure'],
                        'httponly' => (bool) $params['httponly'],
                        'samesite' => $params['samesite'] ?: 'Lax',
                    ]
                );
            }

            session_destroy();
        }
    }

    /** @return array<string, mixed> */
    public function me(): array
    {
        $userId = (int) ($_SESSION['user_id'] ?? 0);
        if ($userId <= 0) {
            throw new HttpException('Bạn chưa đăng nhập', 401);
        }

        $user = $this->repository->findById($userId);
        if ($user === null || (int) $user['is_active'] !== 1) {
            $this->logout();
            throw new HttpException('Phiên đăng nhập không hợp lệ hoặc tài khoản đã bị khóa', 401);
        }

        // Đồng bộ lại role mới nhất từ DB vào session
        $_SESSION['role'] = (string) $user['role'];

        return $this->formatUser($user);
    }

    /** @param array<string, mixed> $user */
    private function startSessionForUser(array $user): void
    {
        if (session_status() === PHP_SESSION_ACTIVE && !headers_sent()) {
            session_regenerate_id(true);
        }

        $_SESSION['user_id'] = (int) $user['id'];
        $_SESSION['role'] = (string) $user['role'];
        $_SESSION['email'] = (string) $user['email'];
    }

    /**
     * @param array<string, mixed> $row
     * @return array<string, mixed>
     */
    private function formatUser(array $row): array
    {
        return [
            'id' => (int) $row['id'],
            'full_name' => (string) $row['full_name'],
            'email' => (string) $row['email'],
            'phone' => $row['phone'] !== null ? (string) $row['phone'] : null,
            'role' => (string) $row['role'],
            'student_verified' => (bool) ((int) $row['student_verified']),
            'is_active' => (bool) ((int) $row['is_active']),
            'created_at' => (string) $row['created_at'],
        ];
    }

     * @return array{user_id: int, role: string}
     */
    private function requireRole(array $allowedRoles): array
    {
        $auth = $this->requireAuth();
        if (!in_array($auth['role'], $allowedRoles, true)) {
            throw new HttpException('Bạn không có quyền thực hiện thao tác này', 403);
        }

        return $auth;
    }

    private function parsePositiveInt(mixed $value, string $field): int
    {
        $parsed = filter_var($value, FILTER_VALIDATE_INT);
        if ($parsed === false || $parsed <= 0) {
            throw new HttpException("Tham số {$field} phải là số nguyên dương", 400);
        }

        return (int) $parsed;
    }

    /** @return array<string, mixed> */
    public function proposeSchedule(int $roomId, int $actorUserId, string $actorRole, string $newStartRaw): array
    {
        $this->requireActiveUser($actorUserId, ['PLAYER', 'ADMIN']);

        $newStartAt = DateTimeImmutable::createFromFormat('Y-m-d H:i:s', trim($newStartRaw));
        if ($newStartAt === false) {
            $newStartAt = DateTimeImmutable::createFromFormat(DateTimeImmutable::ATOM, trim($newStartRaw));
        }
        if ($newStartAt === false) {
            throw new BadRequestException('Định dạng new_start_at không hợp lệ (dùng YYYY-MM-DD HH:MM:SS)');
        }
        if (!in_array((int) $newStartAt->format('i'), [0, 30], true) || (int) $newStartAt->format('s') !== 0) {
            throw new BadRequestException('Giờ đề xuất mới phải là mốc chẵn :00 hoặc :30');
        }

        $now = new DateTimeImmutable();
        if ($newStartAt <= $now) {
            throw new BadRequestException('Giờ đề xuất mới phải ở thời điểm tương lai');
        }

        $lockSql = $this->getRowLockSql();
        $this->pdo->beginTransaction();
        try {
            $stmt = $this->pdo->prepare("SELECT * FROM game_rooms WHERE id = :id{$lockSql}");
            $stmt->execute([':id' => $roomId]);
            $room = $stmt->fetch();

            if ($room === false) {
                throw new NotFoundException('Không tìm thấy phòng chơi');
            }

            $isAdmin = strtoupper($actorRole) === 'ADMIN';
            $isHost = (int) $room['host_user_id'] === $actorUserId;
            if (!$isAdmin && !$isHost) {
                throw new HttpException('Chỉ chủ phòng hoặc quản trị viên mới có quyền đề xuất đổi lịch', 403);
            }

            if (in_array((string) $room['status'], ['CANCELLED', 'COMPLETED', 'EXPIRED'], true)) {
                throw new HttpException('Phòng đã đóng, không thể đề xuất đổi lịch', 409);
            }

            $expiresAt = $now->add(new DateInterval('PT' . self::PROPOSAL_EXPIRE_HOURS . 'H'))->format('Y-m-d H:i:s');
            $ins = $this->pdo->prepare(
                "INSERT INTO schedule_proposals (room_id, proposed_by, old_start_at, new_start_at, expires_at, status)
                 VALUES (:room_id, :proposed_by, :old_start_at, :new_start_at, :expires_at, 'PROPOSED')"
            );
            $ins->execute([
                ':room_id' => $roomId,
                ':proposed_by' => $actorUserId,
                ':old_start_at' => (string) $room['scheduled_start_at'],
                ':new_start_at' => $newStartAt->format('Y-m-d H:i:s'),
                ':expires_at' => $expiresAt,
            ]);
            $proposalId = (int) $this->pdo->lastInsertId();

            $this->pdo->commit();
        } catch (\Throwable $e) {
            if ($this->pdo->inTransaction()) {
                $this->pdo->rollBack();
            }
            throw $e;
        }

        return [
            'id' => $proposalId,
            'room_id' => $roomId,
            'proposed_by' => $actorUserId,
            'old_start_at' => (string) $room['scheduled_start_at'],
            'new_start_at' => $newStartAt->format('Y-m-d H:i:s'),
            'expires_at' => $expiresAt,
            'status' => 'PROPOSED',
        ];
    }

    /** @return array<string, mixed> */
    public function respondScheduleProposal(int $proposalId, int $actorUserId, string $actorRole, string $decision): array
    {
        $this->requireActiveUser($actorUserId, ['PLAYER', 'ADMIN']);
        $normalized = strtoupper(trim($decision));
        if (!in_array($normalized, ['APPROVED', 'REJECTED'], true)) {
            throw new BadRequestException('Quyết định chỉ chấp nhận APPROVED hoặc REJECTED');
        }

        $lockSql = $this->getRowLockSql();
        $now = new DateTimeImmutable();

        $this->pdo->beginTransaction();
        try {
            $pStmt = $this->pdo->prepare("SELECT * FROM schedule_proposals WHERE id = :id{$lockSql}");
            $pStmt->execute([':id' => $proposalId]);
            $proposal = $pStmt->fetch();

            if ($proposal === false) {
                throw new NotFoundException('Không tìm thấy đề xuất đổi lịch');
            }
            if ((string) $proposal['status'] !== 'PROPOSED') {
                throw new HttpException('Đề xuất đổi lịch này đã được xử lý trước đó', 409);
            }

            $expiresAt = new DateTimeImmutable((string) $proposal['expires_at']);
            if ($now > $expiresAt) {
                $exp = $this->pdo->prepare("UPDATE schedule_proposals SET status = 'EXPIRED' WHERE id = :id");
                $exp->execute([':id' => $proposalId]);
                $this->pdo->commit();
                throw new HttpException('Đề xuất đổi lịch đã quá hạn biểu quyết', 409);
            }

            $roomId = (int) $proposal['room_id'];
            $rStmt = $this->pdo->prepare("SELECT * FROM game_rooms WHERE id = :id{$lockSql}");
            $rStmt->execute([':id' => $roomId]);
            $room = $rStmt->fetch();
            if ($room === false) {
                throw new NotFoundException('Không tìm thấy phòng chơi');
            }

            $isAdmin = strtoupper($actorRole) === 'ADMIN';
            $isHost = (int) $room['host_user_id'] === $actorUserId;
            if (!$isAdmin && !$isHost) {
                throw new HttpException('Chỉ chủ phòng hoặc quản trị viên mới có quyền chốt đề xuất đổi lịch', 403);
            }

            if ($normalized === 'REJECTED') {
                $upd = $this->pdo->prepare("UPDATE schedule_proposals SET status = 'REJECTED' WHERE id = :id");
                $upd->execute([':id' => $proposalId]);
                $this->pdo->commit();

                return [
                    'proposal_id' => $proposalId,
                    'room_id' => $roomId,
                    'status' => 'REJECTED',
                ];
            }

            $newStartStr = (string) $proposal['new_start_at'];
            $rule = json_decode((string) $room['snapshot_game_rule'], true);
            $durationMinutes = is_array($rule) ? (int) ($rule['duration_minutes'] ?? 180) : 180;
            $newEndStr = (new DateTimeImmutable($newStartStr))
                ->add(new DateInterval('PT' . $durationMinutes . 'M'))
                ->format('Y-m-d H:i:s');

            $sessStmt = $this->pdo->prepare(
                "SELECT id, dm_user_id, physical_room_id FROM game_sessions
                 WHERE room_id = :room_id AND status = 'SCHEDULED'
                 LIMIT 1{$lockSql}"
            );
            $sessStmt->execute([':room_id' => $roomId]);
            $session = $sessStmt->fetch();

            if (is_array($session)) {
                $conflictStmt = $this->pdo->prepare(
                    "SELECT id FROM game_sessions
                     WHERE room_id != :room_id
                       AND status = 'SCHEDULED'
                       AND (physical_room_id = :pr_id OR dm_user_id = :dm_id)
                       AND start_at < :end_at
                       AND end_at > :start_at
                     LIMIT 1{$lockSql}"
                );
                $conflictStmt->execute([
                    ':room_id' => $roomId,
                    ':pr_id' => (int) $session['physical_room_id'],
                    ':dm_id' => (int) $session['dm_user_id'],
                    ':start_at' => $newStartStr,
                    ':end_at' => $newEndStr,
                ]);
                if ($conflictStmt->fetch() !== false) {
                    throw new HttpException('Khung giờ mới bị trùng lịch DM hoặc phòng vật lý', 409);
                }

                $updSess = $this->pdo->prepare(
                    'UPDATE game_sessions SET start_at = :start_at, end_at = :end_at WHERE id = :id'
                );
                $updSess->execute([
                    ':start_at' => $newStartStr,
                    ':end_at' => $newEndStr,
                    ':id' => (int) $session['id'],
                ]);
            }

            $updRoom = $this->pdo->prepare('UPDATE game_rooms SET scheduled_start_at = :start_at WHERE id = :id');
            $updRoom->execute([':start_at' => $newStartStr, ':id' => $roomId]);

            $updProp = $this->pdo->prepare("UPDATE schedule_proposals SET status = 'APPROVED' WHERE id = :id");
            $updProp->execute([':id' => $proposalId]);

            $this->pdo->commit();
        } catch (\Throwable $e) {
            if ($this->pdo->inTransaction()) {
                $this->pdo->rollBack();
            }
            throw $e;
        }

        return [
            'proposal_id' => $proposalId,
            'status' => 'APPROVED',
            'room' => $this->roomService->getRoomDetail($roomId),
        ];
    }

    /**
     * @param string[]|null $allowedRoles
     * @return array<string, mixed>
     */
    private function requireActiveUser(int $userId, ?array $allowedRoles = null): array
    {
        $stmt = $this->pdo->prepare('SELECT id, role, is_active FROM users WHERE id = :id LIMIT 1');
        $stmt->execute([':id' => $userId]);
        $user = $stmt->fetch();

        if ($user === false || (int) $user['is_active'] !== 1) {
            throw new HttpException('Tài khoản không tồn tại hoặc đã bị khóa', 403);
        }

        if ($allowedRoles !== null) {
            $role = strtoupper((string) $user['role']);
            if (!in_array($role, $allowedRoles, true)) {
                throw new HttpException('Tài khoản không có quyền thực hiện thao tác này', 403);
            }
        }

        return $user;
    }

    private function getRowLockSql(): string
    {
        return $this->pdo->getAttribute(PDO::ATTR_DRIVER_NAME) === 'mysql' ? ' FOR UPDATE' : '';
    }

             WHERE room_id = :room_id AND status IN ('JOINED', 'DEPOSITED', 'PAID_FULL')"

        string $body,
        ?array $payload = null
    ): array {

            "INSERT INTO notifications (user_id, event_type, title, body, payload, is_read)
             VALUES (:user_id, :event_type, :title, :body, :payload, 0)"
        );
        $stmt->execute([
            ':user_id' => $userId,
            ':event_type' => strtoupper(trim($eventType)),
            ':title' => mb_substr(trim($title), 0, 180),
            ':body' => trim($body),
            ':payload' => $payload !== null ? json_encode($payload, JSON_UNESCAPED_UNICODE) : null,
        ]);

        $id = (int) $this->pdo->lastInsertId();
        return $this->fetchByIdOrFail($id);
    }

    /** @return array<int, array<string, mixed>> */
    public function listForUser(int $userId, bool $unreadOnly = false, int $limit = 50): array
    {
        $limit = max(1, min(100, $limit));
        $sql = 'SELECT * FROM notifications WHERE user_id = :user_id';
        if ($unreadOnly) {
            $sql .= ' AND is_read = 0';
        }
        $sql .= " ORDER BY id DESC LIMIT {$limit}";

        $stmt = $this->pdo->prepare($sql);
        $stmt->execute([':user_id' => $userId]);

        return array_map(fn (array $row): array => $this->formatNotification($row), $stmt->fetchAll());
    }

    /** @return array<string, mixed> */
    public function markAsRead(int $notificationId, int $userId): array
    {
        $stmt = $this->pdo->prepare('SELECT * FROM notifications WHERE id = :id LIMIT 1');
        $stmt->execute([':id' => $notificationId]);
        $row = $stmt->fetch();

        if ($row === false) {
            throw new NotFoundException('Không tìm thấy thông báo');
        }
        if ((int) $row['user_id'] !== $userId) {
            throw new HttpException('Bạn không có quyền thao tác trên thông báo của người khác', 403);
        }

        $upd = $this->pdo->prepare('UPDATE notifications SET is_read = 1 WHERE id = :id');
        $upd->execute([':id' => $notificationId]);

        return $this->fetchByIdOrFail($notificationId);
    }

    public function markAllAsRead(int $userId): int
    {
        $upd = $this->pdo->prepare('UPDATE notifications SET is_read = 1 WHERE user_id = :user_id AND is_read = 0');
        $upd->execute([':user_id' => $userId]);

        return $upd->rowCount();
    }

    private function fetchByIdOrFail(int $id): array
    {
        $stmt = $this->pdo->prepare('SELECT * FROM notifications WHERE id = :id LIMIT 1');

            throw new NotFoundException('Không tìm thấy thông báo');
        }

        return $this->formatNotification($row);
    }

    private function formatNotification(array $row): array

    /** @param array<string, mixed> $body */
    private function extractIdempotencyKey(Request $request, array $body): ?string
    {
        $headerKey = $request->getHeader('Idempotency-Key');
        if (is_string($headerKey) && trim($headerKey) !== '') {
            return trim($headerKey);
        }
        if (isset($body['idempotency_key']) && is_string($body['idempotency_key']) && trim($body['idempotency_key']) !== '') {
            return trim($body['idempotency_key']);
        }

        return null;
    }

    /** @return array{user_id: int, role: string} */
    private function requireAuth(): array
    {
        $userId = (int) ($_SESSION['user_id'] ?? 0);
        $role = strtoupper((string) ($_SESSION['role'] ?? ''));

        if ($userId <= 0 || $role === '') {
            throw new HttpException('Bạn cần đăng nhập để thực hiện thao tác này', 401);
        }

        return ['user_id' => $userId, 'role' => $role];
    }

    /**
     * @param string[] $allowedRoles
     * @return array{user_id: int, role: string}
     */
    private function requireRole(array $allowedRoles): array
    {
        $auth = $this->requireAuth();
        if (!in_array($auth['role'], $allowedRoles, true)) {
            throw new HttpException('Bạn không có quyền thực hiện thao tác này', 403);
        }

        return $auth;
    }

    private function parsePositiveInt(mixed $value, string $field): int
    {
        $parsed = filter_var($value, FILTER_VALIDATE_INT);
        if ($parsed === false || $parsed <= 0) {
            throw new HttpException("Tham số {$field} phải là số nguyên dương", 400);
        }

        return (int) $parsed;
    }

            if ($refund === false) {
                throw new NotFoundException('Không tìm thấy yêu cầu hoàn tiền');

                throw new HttpException('Yêu cầu hoàn tiền này đã được xử lý trước đó', 409);

            $updRef = $this->pdo->prepare(
                'UPDATE refunds SET status = :status, reference_code = :ref_code, processed_by = :processed_by WHERE id = :id'
            );
            $updRef->execute([
                ':status' => $normalized,
                ':ref_code' => $refCode,
                ':processed_by' => $actorUserId,
                ':id' => $refundId,
            ]);

            if ($normalized === 'PROCESSED') {
                $updPay = $this->pdo->prepare("UPDATE payment_transactions SET status = 'REFUNDED' WHERE id = :id");
                $updPay->execute([':id' => (int) $refund['payment_id']]);
            }

            $this->pdo->commit();
        } catch (\Throwable $e) {
            if ($this->pdo->inTransaction()) {
                $this->pdo->rollBack();

    /**
     * @param string[]|null $allowedRoles
     * @return array<string, mixed>

    private function requireActiveUser(int $userId, ?array $allowedRoles = null): array
    {
        $stmt = $this->pdo->prepare('SELECT id, role, is_active FROM users WHERE id = :id LIMIT 1');
        $stmt->execute([':id' => $userId]);
        $user = $stmt->fetch();

        if ($user === false || (int) $user['is_active'] !== 1) {
            throw new HttpException('Tài khoản không tồn tại hoặc đã bị khóa', 403);
        }

        if ($allowedRoles !== null) {
            $role = strtoupper((string) $user['role']);
            if (!in_array($role, $allowedRoles, true)) {
                throw new HttpException('Tài khoản không có quyền thực hiện thao tác này', 403);
            }
        }

        return $user;
    }

    private function getRowLockSql(): string
    {
        return $this->pdo->getAttribute(PDO::ATTR_DRIVER_NAME) === 'mysql' ? ' FOR UPDATE' : '';
    }

<?php

declare(strict_types=1);

namespace App\Infrastructure\Config;

class Config
{
    /** @var array<string, mixed> */
    private static array $repository = [];
    private static bool $loaded = false;

    public static function load(?string $basePath = null): void
    {
        if (self::$loaded && $basePath === null) {
            return;
        }

        $basePath = $basePath ?? dirname(__DIR__, 3);
        $envFile = rtrim($basePath, '/\\') . DIRECTORY_SEPARATOR . '.env';

        if (file_exists($envFile)) {
            $lines = file($envFile, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES);
            if ($lines !== false) {
                foreach ($lines as $line) {
                    $line = trim($line);
                    if ($line === '' || str_starts_with($line, '#')) {
                        continue;
                    }

                    if (str_contains($line, '=')) {
                        [$name, $value] = explode('=', $line, 2);
                        $name = trim($name);
                        $value = trim($value);

                        if ((str_starts_with($value, '"') && str_ends_with($value, '"')) ||
                            (str_starts_with($value, "'") && str_ends_with($value, "'"))) {
                            $value = substr($value, 1, -1);
                        }

                        if (!array_key_exists($name, self::$repository)) {
                            self::$repository[$name] = $value;
                        }
                        if (!array_key_exists($name, $_ENV)) {
                            $_ENV[$name] = $value;
                        }
                    }
                }
            }
        }

        self::$loaded = true;
    }

    public static function get(string $key, mixed $default = null): mixed
    {
        if (!self::$loaded) {
            self::load();
        }

        if (array_key_exists($key, self::$repository)) {
            return self::castValue(self::$repository[$key]);
        }

        $envVal = getenv($key);
        if ($envVal !== false) {
            return self::castValue($envVal);
        }

        if (isset($_ENV[$key])) {
            return self::castValue($_ENV[$key]);
        }

        return $default;
    }

    public static function set(string $key, mixed $value): void
    {
        self::$repository[$key] = $value;
    }

    public static function reset(): void
    {
        self::$repository = [];
        self::$loaded = false;
    }

    private static function castValue(mixed $value): mixed
    {
        if (!is_string($value)) {
            return $value;
        }

        return match (strtolower($value)) {
            'true', '(true)' => true,
            'false', '(false)' => false,
            'null', '(null)' => null,
            'empty', '(empty)' => '',
            default => $value,
        };
    }
}

class ConnectionFactory

    /**
     * Create a new PDO instance from environment configuration or explicit options.
     *
     * @param array<string, mixed> $overrideConfig
     * @throws DatabaseException
     */
    public static function create(array $overrideConfig = []): PDO
    {
        $driver = (string) ($overrideConfig['driver'] ?? Config::get('DB_CONNECTION', 'mysql'));
        $host = (string) ($overrideConfig['host'] ?? Config::get('DB_HOST', '127.0.0.1'));
        $port = (int) ($overrideConfig['port'] ?? Config::get('DB_PORT', 3306));
        $database = (string) ($overrideConfig['database'] ?? Config::get('DB_DATABASE', 'booking_v16'));
        $username = (string) ($overrideConfig['username'] ?? Config::get('DB_USERNAME', 'root'));
        $password = (string) ($overrideConfig['password'] ?? Config::get('DB_PASSWORD', ''));
        $charset = (string) ($overrideConfig['charset'] ?? Config::get('DB_CHARSET', 'utf8mb4'));

        $dsn = sprintf('%s:host=%s;port=%d;dbname=%s;charset=%s', $driver, $host, $port, $database, $charset);

        $defaultOptions = [
            PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES => false,
            PDO::ATTR_PERSISTENT => false,
        ];

        // Ensure proper collation on MySQL
        if ($driver === 'mysql') {
            $defaultOptions[PDO::MYSQL_ATTR_INIT_COMMAND] = "SET NAMES '{$charset}' COLLATE 'utf8mb4_unicode_ci'";
        }

        try {
            return new PDO($dsn, $username, $password, $defaultOptions);
        } catch (PDOException $e) {
            $sanitizedMsg = ExceptionHandler::sanitize($e->getMessage());

            $errorInfo = sprintf(
                'Failed to connect to database [%s] on host [%s:%d]: %s',
                $database,
                $host,
                $port,
                $sanitizedMsg
            );

            // Raw PDOException is deliberately NOT chained as previous to prevent leaking
            // plain-text password from PDO::__construct() stack trace frame arguments.
            throw new DatabaseException($errorInfo, (int) $e->getCode());
        }
    }
}

class DatabaseConnection
{
    private static ?PDO $instance = null;

    /**
     * Get or initialize the shared PDO connection.
     */
    public static function getInstance(array $config = []): PDO
    {
        if (self::$instance === null || !empty($config)) {
            self::$instance = ConnectionFactory::create($config);
        }

        return self::$instance;
    }

    /**
     * Set a custom PDO instance (useful for mocks or test SQLite).
     */
    public static function setInstance(?PDO $pdo): void
    {
        self::$instance = $pdo;
    }

    /**
     * Verify database connectivity by running a lightweight query.
     */
    public static function checkConnectivity(?PDO $pdo = null): bool
    {
        try {
            $connection = $pdo ?? self::getInstance();
            $statement = $connection->query('SELECT 1');
            return $statement !== false && (int) $statement->fetchColumn() === 1;
        } catch (Throwable) {
            return false;
        }
    }

    /**
     * Execute a callable within a transactional boundary.
     *
     * @template T
     * @param callable(PDO): T $callback
     * @return T
     * @throws Throwable
     */
    public static function transaction(callable $callback, ?PDO $pdo = null): mixed
    {
        $connection = $pdo ?? self::getInstance();

        if ($connection->inTransaction()) {
            return $callback($connection);
        }

        $connection->beginTransaction();

        try {
            $result = $callback($connection);
            $connection->commit();
            return $result;
        } catch (Throwable $e) {
            if ($connection->inTransaction()) {
                $connection->rollBack();
            }
            throw $e;
        }
    }
}

    room_code VARCHAR(20) NOT NULL UNIQUE,
    game_id BIGINT UNSIGNED NOT NULL,
    host_user_id BIGINT UNSIGNED NOT NULL,
    scheduled_start_at DATETIME NOT NULL COMMENT 'Chi nhan phut :00 hoac :30',
    status VARCHAR(40) NOT NULL DEFAULT 'FORMING',
    formation_deadline_at DATETIME NOT NULL COMMENT 'Han 4 gio gom nguoi',
    reopen_count TINYINT UNSIGNED NOT NULL DEFAULT 0,
    snapshot_game_rule JSON NOT NULL,
    cancel_reason TEXT NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_rooms_game FOREIGN KEY (game_id) REFERENCES games(id),
    CONSTRAINT fk_rooms_host FOREIGN KEY (host_user_id) REFERENCES users(id),
    INDEX idx_rooms_status_start (status, scheduled_start_at),
    INDEX idx_rooms_host_status (host_user_id, status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 4. Thành viên phòng chơi (RoomParticipants - bản ghi terminal không tái kích hoạt, rejoin tạo dòng mới)
CREATE TABLE IF NOT EXISTS room_participants (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    room_id BIGINT UNSIGNED NOT NULL,
    user_id BIGINT UNSIGNED NOT NULL,
    status VARCHAR(40) NOT NULL DEFAULT 'PENDING_DEPOSIT',
    deposit_deadline_at DATETIME NULL COMMENT 'Han coc 2 gio rieng tung nguoi',
    joined_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    left_at DATETIME NULL,
    leave_reason TEXT NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_part_room FOREIGN KEY (room_id) REFERENCES game_rooms(id),
    CONSTRAINT fk_part_user FOREIGN KEY (user_id) REFERENCES users(id),
    INDEX idx_part_room_status (room_id, status),
    INDEX idx_part_user_status (user_id, status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 5. Phòng vật lý tại quán
CREATE TABLE IF NOT EXISTS physical_rooms (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    capacity TINYINT UNSIGNED NOT NULL COMMENT 'Tinh ca DM',
    theme VARCHAR(100) NULL,
    status VARCHAR(30) NOT NULL DEFAULT 'AVAILABLE',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 6. Phân công Quản trò (DM Assignments)
CREATE TABLE IF NOT EXISTS dm_assignments (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    room_id BIGINT UNSIGNED NOT NULL,
    dm_user_id BIGINT UNSIGNED NOT NULL,
    assigned_by BIGINT UNSIGNED NOT NULL,
    status VARCHAR(30) NOT NULL DEFAULT 'PENDING',
    reason TEXT NULL,
    responded_at DATETIME NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_dm_room FOREIGN KEY (room_id) REFERENCES game_rooms(id),
    CONSTRAINT fk_dm_user FOREIGN KEY (dm_user_id) REFERENCES users(id),
    CONSTRAINT fk_dm_admin FOREIGN KEY (assigned_by) REFERENCES users(id),
    INDEX idx_dm_room_status (room_id, status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 7. Đề xuất đổi lịch (Schedule Proposals)
CREATE TABLE IF NOT EXISTS schedule_proposals (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    room_id BIGINT UNSIGNED NOT NULL,
    proposed_by BIGINT UNSIGNED NOT NULL,
    old_start_at DATETIME NOT NULL,
    new_start_at DATETIME NOT NULL,
    expires_at DATETIME NOT NULL COMMENT 'Han bieu quyet 2 gio',
    status VARCHAR(30) NOT NULL DEFAULT 'PROPOSED',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_prop_room FOREIGN KEY (room_id) REFERENCES game_rooms(id),
    CONSTRAINT fk_prop_user FOREIGN KEY (proposed_by) REFERENCES users(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 8. Phiên chơi chính thức (Game Sessions)
CREATE TABLE IF NOT EXISTS game_sessions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    room_id BIGINT UNSIGNED NOT NULL UNIQUE,
    dm_user_id BIGINT UNSIGNED NOT NULL,
    physical_room_id BIGINT UNSIGNED NOT NULL,
    start_at DATETIME NOT NULL,
    end_at DATETIME NOT NULL,
    status VARCHAR(30) NOT NULL DEFAULT 'SCHEDULED',
    cancel_reason TEXT NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_sess_room FOREIGN KEY (room_id) REFERENCES game_rooms(id),
    CONSTRAINT fk_sess_dm FOREIGN KEY (dm_user_id) REFERENCES users(id),
    CONSTRAINT fk_sess_phys FOREIGN KEY (physical_room_id) REFERENCES physical_rooms(id),
    INDEX idx_sess_phys_time (physical_room_id, start_at, end_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 9. Giao dịch thanh toán (Player Payments - hỗ trợ Idempotency)
CREATE TABLE IF NOT EXISTS payment_transactions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    room_id BIGINT UNSIGNED NOT NULL,
    participant_id BIGINT UNSIGNED NOT NULL,
    user_id BIGINT UNSIGNED NOT NULL,
    idempotency_key VARCHAR(100) NULL UNIQUE,
    payment_type VARCHAR(30) NOT NULL COMMENT 'DEPOSIT | REMAINING',
    amount_vnd BIGINT UNSIGNED NOT NULL,
    status VARCHAR(30) NOT NULL DEFAULT 'DEPOSIT_PENDING',
    confirmed_by BIGINT UNSIGNED NULL,
    note VARCHAR(255) NULL,
    paid_at DATETIME NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_pay_room FOREIGN KEY (room_id) REFERENCES game_rooms(id),
    CONSTRAINT fk_pay_part FOREIGN KEY (participant_id) REFERENCES room_participants(id),
    CONSTRAINT fk_pay_user FOREIGN KEY (user_id) REFERENCES users(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 10. Hoàn tiền (Refunds)
CREATE TABLE IF NOT EXISTS refunds (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    payment_id BIGINT UNSIGNED NOT NULL,
    participant_id BIGINT UNSIGNED NOT NULL,
    amount_vnd BIGINT UNSIGNED NOT NULL,
    status VARCHAR(30) NOT NULL DEFAULT 'REQUESTED',
    reference_code VARCHAR(100) NULL,
    reason TEXT NOT NULL,
    processed_by BIGINT UNSIGNED NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_ref_pay FOREIGN KEY (payment_id) REFERENCES payment_transactions(id),
    CONSTRAINT fk_ref_part FOREIGN KEY (participant_id) REFERENCES room_participants(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 11. Thông báo (Notifications)
CREATE TABLE IF NOT EXISTS notifications (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    event_type VARCHAR(60) NOT NULL,
    title VARCHAR(180) NOT NULL,
    body TEXT NOT NULL,
    payload JSON NULL,
    is_read TINYINT(1) NOT NULL DEFAULT 0,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_notif_user FOREIGN KEY (user_id) REFERENCES users(id),
    INDEX idx_notif_user_read (user_id, is_read, created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 12. Nhật ký kiểm toán (Audit Logs)
CREATE TABLE IF NOT EXISTS audit_logs (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    actor_user_id BIGINT UNSIGNED NULL,
    action VARCHAR(80) NOT NULL,
    target_type VARCHAR(50) NOT NULL,
    target_id BIGINT UNSIGNED NOT NULL,
    reason TEXT NULL,
    metadata JSON NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_audit_target (target_type, target_id),
    INDEX idx_audit_actor (actor_user_id, created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
