import { authState } from './api/auth-state.js';
import { api } from './api/api-client.js';
import { toast } from './utils/toast.js';
import { escapeHtml } from './utils/sanitize.js';
import { AuthModal } from './pages/auth/auth-modal.js';
import { HomeView } from './pages/home/home-view.js';
import { GamesView } from './pages/games/games-view.js';
import { RoomsView } from './pages/rooms/rooms-view.js';
import { DmView } from './pages/dm/dm-view.js';
import { AdminView } from './pages/admin/admin-view.js';
import { NotificationsView } from './pages/notifications/notifications-view.js';

class App {
  constructor() {
    this.mountPoint = document.getElementById('app-mount');
    this.userContainer = document.getElementById('nav-user-container');
    this.notifBtn = document.getElementById('nav-btn-notif');
    this.notifBadge = document.getElementById('notif-unread-count');
    this.mobileDrawer = document.getElementById('mobile-nav-drawer');
    this.hamburgerBtn = document.getElementById('nav-hamburger-btn');
    this.hamburgerIcon = document.getElementById('nav-hamburger-icon');
    this.currentTab = 'home';

    // Initialize Auth Modal
    this.authModal = new AuthModal((user, selectedRole) => {
      this.updateNavigationForRole(user);
      this.renderUserNav();
      this.updateNotificationsCount();
      if (user) {
        if (user.role === 'DM' && (this.currentTab === 'dm' || selectedRole === 'DM')) {
          this.navigateTo('dm');
        } else if (user.role === 'ADMIN' && (this.currentTab === 'admin' || selectedRole === 'ADMIN')) {
          this.navigateTo('admin');
        }
      }
      this.refreshCurrentView();
    });
    this.authModal.render();

    // Initialize Views
    this.views = {
      home: new HomeView(this.mountPoint, {
        onNavigate: (tab) => this.navigateTo(tab),
        onOpenAuth: (role) => this.authModal.show(false, role || 'PLAYER')
      }),
      games: new GamesView(this.mountPoint, () => {
        if (!authState.isAuthenticated()) {
          this.authModal.show();
        } else {
          this.navigateTo('rooms');
        }
      }),
      rooms: new RoomsView(this.mountPoint, () => this.authModal.show()),
      dm: new DmView(this.mountPoint, () => this.authModal.show(false, 'DM')),
      admin: new AdminView(this.mountPoint, () => this.authModal.show(false, 'ADMIN')),
      notifications: new NotificationsView(this.mountPoint)
    };
  }

  async init() {
    this.setupNavigation();
    this.setupAuthStateListener();

    // Check session login state
    try {
      await authState.checkSession();
    } catch {
      // not logged in
    }

    const user = authState.getUser();
    this.updateNavigationForRole(user);
    this.renderUserNav();
    this.updateNotificationsCount();

    // Handle hash change or default
    window.addEventListener('hashchange', () => this.handleRouting());
    this.handleRouting();

    // Periodic check for notifications if logged in
    setInterval(() => {
      if (authState.isAuthenticated()) {
        this.updateNotificationsCount();
      }
    }, 15000);
  }

  setupNavigation() {
    const navButtons = document.querySelectorAll('.nav-tab-btn');
    navButtons.forEach(btn => {
      btn.addEventListener('click', () => {
        const tab = btn.dataset.tab;
        this.navigateTo(tab);
      });
    });

    if (this.notifBtn) {
      this.notifBtn.addEventListener('click', () => {
        this.navigateTo('notifications');
      });
    }

    // Mobile Hamburger Toggle
    if (this.hamburgerBtn && this.mobileDrawer) {
      this.hamburgerBtn.addEventListener('click', () => {
        const isOpen = this.mobileDrawer.classList.toggle('open');
        if (this.hamburgerIcon) {
          this.hamburgerIcon.textContent = isOpen ? '✕' : '☰';
        }
      });

      // Mobile Nav Links
      const mobileLinks = this.mobileDrawer.querySelectorAll('.mobile-nav-link');
      mobileLinks.forEach(link => {
        link.addEventListener('click', (e) => {
          e.preventDefault();
          const tab = link.dataset.tab;
          this.closeMobileDrawer();
          this.navigateTo(tab);
        });
      });
    }
  }

  closeMobileDrawer() {
    if (this.mobileDrawer && this.mobileDrawer.classList.contains('open')) {
      this.mobileDrawer.classList.remove('open');
      if (this.hamburgerIcon) {
        this.hamburgerIcon.textContent = '☰';
      }
    }
  }

  navigateTo(tab) {
    window.location.hash = `#${tab}`;
  }

  setupAuthStateListener() {
    authState.subscribe(() => {
      const user = authState.getUser();
      this.updateNavigationForRole(user);
      this.renderUserNav();
      this.updateNotificationsCount();
      this.refreshCurrentView();
    });
  }

  updateNavigationForRole(user) {
    const dmBtn = document.getElementById('nav-btn-dm');
    const adminBtn = document.getElementById('nav-btn-admin');
    const mobileDm = document.getElementById('mobile-nav-dm');
    const mobileAdmin = document.getElementById('mobile-nav-admin');

    const isDm = Boolean(user && (user.role === 'DM' || user.role === 'ADMIN'));
    const isAdmin = Boolean(user && user.role === 'ADMIN');

    if (dmBtn) dmBtn.style.display = isDm ? 'flex' : 'none';
    if (mobileDm) mobileDm.style.display = isDm ? 'flex' : 'none';

    if (adminBtn) adminBtn.style.display = isAdmin ? 'flex' : 'none';
    if (mobileAdmin) mobileAdmin.style.display = isAdmin ? 'flex' : 'none';
  }

  handleRouting() {
    let hash = window.location.hash.replace('#', '') || 'home';
    const user = authState.getUser();

    // Role-specific navigation guards
    if (hash === 'dm') {
      if (!user || (user.role !== 'DM' && user.role !== 'ADMIN')) {
        toast.warning('Khu vực dành cho Quản Trò (DM). Vui lòng đăng nhập với vai trò DM.');
        this.navigateTo('home');
        this.authModal.show(false, 'DM');
        return;
      }
    } else if (hash === 'admin') {
      if (!user || user.role !== 'ADMIN') {
        toast.warning('Khu vực dành cho Quản Trị Viên (Admin). Vui lòng đăng nhập với tài khoản Admin.');
        this.navigateTo('home');
        this.authModal.show(false, 'ADMIN');
        return;
      }
    }

    if (!this.views[hash]) {
      hash = 'home';
    }
    this.currentTab = hash;
    this.closeMobileDrawer();

    // Update active nav button (Desktop)
    document.querySelectorAll('.nav-tab-btn').forEach(btn => {
      if (btn.dataset.tab === hash) {
        btn.classList.add('active');
      } else {
        btn.classList.remove('active');
      }
    });

    // Update active nav link (Mobile)
    if (this.mobileDrawer) {
      this.mobileDrawer.querySelectorAll('.mobile-nav-link').forEach(link => {
        if (link.dataset.tab === hash) {
          link.classList.add('active');
        } else {
          link.classList.remove('active');
        }
      });
    }

    this.refreshCurrentView();
  }

  async refreshCurrentView() {
    const view = this.views[this.currentTab];
    if (view && typeof view.render === 'function') {
      await view.render();
    }
  }

  renderUserNav() {
    if (!this.userContainer) return;

    const user = authState.getUser();

    if (user) {
      let roleBadgeColor = 'var(--color-primary)';
      if (user.role === 'ADMIN') roleBadgeColor = 'var(--color-error)';
      else if (user.role === 'DM') roleBadgeColor = 'var(--color-warning)';

      this.userContainer.innerHTML = `
        <div style="display: flex; align-items: center; gap: 10px;">
          <div class="user-pill" style="cursor: pointer;" id="btn-switch-user" title="Nhấn để đổi nhanh tài khoản khác">
            <span class="badge" style="background: ${roleBadgeColor}; color: white; font-size: 0.7rem;">
              ${escapeHtml(user.role)}
            </span>
            <strong style="color: var(--color-text-main); font-size: 0.85rem;">${escapeHtml(user.name)}</strong>
            <span style="font-size: 0.7rem; color: var(--color-text-muted);">🔄 Đổi vai trò</span>
          </div>
          <button id="btn-logout" class="btn" style="padding: 6px 12px; font-size: 0.8rem; background: transparent; border: 1px solid var(--color-border); color: var(--color-text-muted);" title="Đăng xuất khỏi hệ thống">
            Đăng xuất
          </button>
        </div>
      `;

      this.userContainer.querySelector('#btn-switch-user').onclick = () => {
        this.authModal.show();
      };

      this.userContainer.querySelector('#btn-logout').onclick = async () => {
        try {
          await authState.logout();
          this.updateNavigationForRole(null);
          toast.info('Đã đăng xuất!');
          if (this.currentTab === 'dm' || this.currentTab === 'admin') {
            this.navigateTo('home');
          }
        } catch (err) {
          toast.error(err.message || 'Lỗi đăng xuất');
        }
      };
    } else {
      this.userContainer.innerHTML = `
        <button id="btn-open-login" class="btn btn-primary" style="padding: 7px 16px; font-size: 0.85rem;">
          Đăng nhập / Đổi vai trò
        </button>
      `;

      this.userContainer.querySelector('#btn-open-login').onclick = () => {
        this.authModal.show();
      };
    }
  }

  async updateNotificationsCount() {
    if (!this.notifBadge) return;

    if (!authState.isAuthenticated()) {
      this.notifBadge.style.display = 'none';
      return;
    }

    try {
      const res = await api.get('/notifications?unread_only=1');
      const unreadList = res.data || [];
      const count = unreadList.length;

      if (count > 0) {
        this.notifBadge.textContent = count > 99 ? '99+' : count;
        this.notifBadge.style.display = 'block';
      } else {
        this.notifBadge.style.display = 'none';
      }
    } catch {
      this.notifBadge.style.display = 'none';
    }
  }
}

document.addEventListener('DOMContentLoaded', () => {
  const app = new App();
  app.init();
});

import { config } from '../utils/config.js';

export class ApiError extends Error {
  constructor(message, status = 0, code = 'NETWORK_ERROR', details = null) {
    super(message);
    this.name = 'ApiError';
    this.status = status;
    this.code = code;
    this.details = details;
  }
}

/**
 * Standard API Client with automatic CSRF token management and PHP Session support.
 */
export class ApiClient {
  constructor(baseUrl = config.apiBaseUrl) {
    this.baseUrl = (baseUrl || '').replace(/\/+$/, '');
    this.csrfToken = null;
    this.csrfPromise = null;
  }

  /**
   * Clear cached CSRF token (called upon logout or CSRF invalidation)
   */
  clearCsrfToken() {
    this.csrfToken = null;
    this.csrfPromise = null;
  }

  /**
   * Fetch and cache CSRF token from backend
   */
  async ensureCsrfToken() {
    if (this.csrfToken) {
      return this.csrfToken;
    }

    if (!this.csrfPromise) {
      this.csrfPromise = (async () => {
        try {
          const res = await fetch(`${this.baseUrl}/auth/csrf-token`, {
            credentials: 'include',
            headers: { 'Accept': 'application/json' },
          });
          const data = await res.json();
          if (data && data.success && data.data && data.data.csrf_token) {
            this.csrfToken = data.data.csrf_token;
            return this.csrfToken;
          }
        } catch {
          // Ignore prefetch error in non-browser or disconnected environment
        } finally {
          this.csrfPromise = null;
        }
        return null;
      })();
    }

    return this.csrfPromise;
  }

  async request(endpoint, options = {}) {
    const url = `${this.baseUrl}/${endpoint.replace(/^\/+/, '')}`;
    const method = (options.method || 'GET').toUpperCase();

    // Immediate check if caller already aborted
    if (options.signal && options.signal.aborted) {
      const reason = typeof options.signal.reason === 'string' ? options.signal.reason : 'Request aborted by caller';
      throw new ApiError(reason, 0, 'REQUEST_ABORTED');
    }

    // Attach CSRF token for mutations
    if (['POST', 'PUT', 'PATCH', 'DELETE'].includes(method)) {
      await this.ensureCsrfToken();
    }

    const headers = {
      'Accept': 'application/json',
      ...(options.body ? { 'Content-Type': 'application/json' } : {}),
      ...(this.csrfToken ? { 'X-CSRF-Token': this.csrfToken } : {}),
      ...(options.headers || {})
    };

    const controller = new AbortController();
    const timeoutMs = options.timeout || config.defaultTimeoutMs;
    let isTimeout = false;

    // Propagate caller signal abort to internal controller
    let abortListener = null;
    if (options.signal) {
      abortListener = () => {
        controller.abort(options.signal.reason);
      };
      options.signal.addEventListener('abort', abortListener);
    }

    const timer = setTimeout(() => {
      isTimeout = true;
      controller.abort();
    }, timeoutMs);

    try {
      const response = await fetch(url, {
        credentials: 'include', // Always transmit PHP session cookie
        ...options,
        headers,
        signal: controller.signal,
      });

      clearTimeout(timer);

      let data = null;
      const contentType = response.headers.get('content-type') || '';
      if (contentType.includes('application/json')) {
        try {
          data = await response.json();
        } catch {
          throw new ApiError('Failed to parse server response as JSON', response.status, 'INVALID_JSON_RESPONSE');
        }
      } else {
        data = await response.text();
      }

      if (!response.ok) {
        const errorInfo = (typeof data === 'object' && data?.error) ? data.error : null;
        const message = errorInfo?.message || data?.message || `HTTP error ${response.status}`;
        const code = errorInfo?.code || `HTTP_${response.status}`;
        const details = errorInfo?.details || null;

        if (code === 'CSRF_TOKEN_INVALID' || (typeof message === 'string' && message.includes('CSRF'))) {
          this.clearCsrfToken();
        }

        throw new ApiError(message, response.status, code, details);
      }

      return data;
    } catch (err) {
      clearTimeout(timer);

      if (err instanceof ApiError) {
        throw err;
      }

      if (err.name === 'AbortError') {
        if (isTimeout) {
          throw new ApiError(`Request timed out after ${timeoutMs}ms`, 0, 'REQUEST_TIMEOUT');
        }
        const reason = options.signal?.reason;
        const msg = typeof reason === 'string' ? reason : 'Request aborted by caller';
        throw new ApiError(msg, 0, 'REQUEST_ABORTED');
      }

      throw new ApiError('Failed to fetch', 0, 'NETWORK_FAILURE', { originalError: err.message });
    } finally {
      if (options.signal && abortListener) {
        options.signal.removeEventListener('abort', abortListener);
      }
    }
  }

  get(endpoint, options = {}) {
    return this.request(endpoint, { ...options, method: 'GET' });
  }

  post(endpoint, body = {}, options = {}) {
    return this.request(endpoint, {
      ...options,
      method: 'POST',
      body: JSON.stringify(body),
    });
  }

  put(endpoint, body = {}, options = {}) {
    return this.request(endpoint, {
      ...options,
      method: 'PUT',
      body: JSON.stringify(body),
    });
  }

  patch(endpoint, body = {}, options = {}) {
    return this.request(endpoint, {
      ...options,
      method: 'PATCH',
      body: JSON.stringify(body),
    });
  }

  delete(endpoint, options = {}) {
    return this.request(endpoint, { ...options, method: 'DELETE' });
  }
}

import { api } from './api-client.js';

class AuthState {
  constructor() {
    this.user = null;
    this.listeners = [];
  }

  subscribe(listener) {
    this.listeners.push(listener);
    return () => {
      this.listeners = this.listeners.filter(l => l !== listener);
    };
  }

  notify() {
    const normalizedUser = this.getUser();
    this.listeners.forEach(listener => {
      try {
        listener(normalizedUser);
      } catch (e) {
        console.error('[AuthState] Listener error:', e);
      }
    });
  }

  getUser() {
    if (!this.user) return null;
    return {
      ...this.user,
      id: Number(this.user.id),
      name: this.user.full_name || this.user.name || 'Người dùng',
      role: (this.user.role_code || this.user.role || 'PLAYER').toUpperCase(),
    };
  }

  async checkAuth() {
    try {
      const res = await api.get('/auth/me');
      if (res && res.success && res.data && res.data.user) {
        this.user = res.data.user;
      } else {
        this.user = null;
      }
    } catch {
      this.user = null;
    }
    this.notify();
    return this.getUser();
  }

  async checkSession() {
    return this.checkAuth();
  }

  async login(email, password) {
    const res = await api.post('/auth/login', { email, password });
    if (res && res.success && res.data && res.data.user) {
      this.user = res.data.user;
      this.notify();
      return this.getUser();
    }
    throw new Error(res?.message || 'Đăng nhập thất bại');
  }

  async register(data) {
    const res = await api.post('/auth/register', data);
    if (res && res.success && res.data && res.data.user) {
      this.user = res.data.user;
      this.notify();
      return this.getUser();
    }
    throw new Error(res?.message || 'Đăng ký thất bại');
  }

  async logout() {
    try {
      await api.post('/auth/logout');
    } finally {
      api.clearCsrfToken();
      this.user = null;
      this.notify();
    }
  }

  isAuthenticated() {
    return this.user !== null;
  }

  hasRole(role) {
    if (!this.user) return false;
    const userRole = (this.user.role_code || this.user.role || '').toUpperCase();
    return userRole === role.toUpperCase();
  }

  isAdmin() {
    return this.hasRole('ADMIN');
  }

  isDM() {
    return this.hasRole('DM') || this.isAdmin();
  }
}

export const authState = new AuthState();

          if (errorAlert) {
            errorAlert.textContent = errMsg;

          }

        } finally {
          if (submitBtn) submitBtn.disabled = false;
          if (submitText) {
            submitText.textContent = this.isRegisterMode
              ? 'Đăng ký tài khoản'
              : `Đăng nhập (${roleConfig.name})`;
          }
        }
      };
    }

   * Open modal (supports both open() and show() for full backwards compatibility)
   * @param {boolean} registerMode Whether to open in registration mode
   * @param {string} defaultRole Initial selected role ('PLAYER', 'DM', 'ADMIN')
   */
  open(registerMode = false, defaultRole = 'PLAYER') {

    const configs = getRoleConfigs();

  }

   * Alias for open()
   */

    this.open(registerMode, defaultRole);
  }

  close() {
    if (this.modal && this.modal.open) {
      this.modal.close();
    }
  }

  setupRejectDialog() {
    const dialog = this.container.querySelector('#dm-reject-dialog');
    const closeBtn = this.container.querySelector('#btn-close-reject-dialog');
    const cancelBtn = this.container.querySelector('#btn-cancel-reject');

    const closeDialog = () => {
      dialog.close();
      form.reset();
    };

    closeBtn.onclick = closeDialog;
    cancelBtn.onclick = closeDialog;

    form.onsubmit = async (e) => {
      e.preventDefault();
      const assignmentId = this.container.querySelector('#dm-reject-assignment-id').value;
      const reason = this.container.querySelector('#dm-reject-reason').value.trim();

      if (!reason) {
        toast.error('Vui lòng nhập lý do từ chối ca!');
        return;
      }

      try {
        const res = await api.post(`/dm/assignments/${assignmentId}/reject`, { rejection_reason: reason });
        toast.success(res.message || 'Đã từ chối ca thành công!');
        closeDialog();
        this.fetchAssignments();
      } catch (err) {
        toast.error(err.message || 'Không thể từ chối ca');
      }
    };

import { healthApi } from '../../api/health-api.js';

export class HealthPage {
  constructor(mountElement) {
    this.container = mountElement;
    this.state = {
      status: 'idle', // 'idle' | 'loading' | 'success' | 'error'
      data: null,
      error: null,
      latency: null,
    };
  }

  async runCheck() {
    this.state.status = 'loading';
    this.state.error = null;
    this.render();

    const startTime = performance.now();
    try {
      const response = await healthApi.checkHealth();
      const endTime = performance.now();
      this.state.status = 'success';
      this.state.data = response;
      this.state.latency = Math.round(endTime - startTime);
    } catch (err) {
      this.state.status = 'error';
      this.state.error = err;
    } finally {
      this.render();
    }
  }

  render() {
    const { status, data, error, latency } = this.state;

    const section = document.createElement('section');
    section.className = 'card';

    // Card title
    const cardTitle = document.createElement('div');
    cardTitle.className = 'card-title';
    const titleSpan = document.createElement('span');

    cardTitle.appendChild(titleSpan);

    // Status badge
    const badge = document.createElement('span');
    const dot = document.createElement('span');

    if (status === 'loading') {

    } else if (status === 'success') {

      p.textContent = 'Contacting backend at /api/health...';
      bodyContainer.appendChild(p);
    } else if (status === 'success') {
      const healthData = data?.data || {};

      const dataGrid = document.createElement('div');

      const createCell = (label, value) => {

    section.appendChild(bodyContainer);

    this.container.innerHTML = `<div style="text-align: center; padding: 60px; color: var(--color-text-muted);">Đang tải thông tin phòng chơi...</div>`;

    try {
      const res = await api.get(`/rooms/${roomId}`);
      this.selectedRoom = res?.data?.room;

      if (!this.selectedRoom) {
        throw new Error('Không tìm thấy thông tin phòng');
      }

      this.renderRoomDetail();
    } catch (err) {
      toast.error(err.message);
      this.render();
    }
  }

  bindProposalEvents() {
    const proposeBtn = this.container.querySelector('#btn-open-propose-modal');
    if (proposeBtn) {
      proposeBtn.onclick = () => this.showProposeModal();
    }

    this.container.querySelectorAll('.btn-vote').forEach(btn => {
      btn.onclick = async () => {
        const proposalId = btn.dataset.id;
        const vote = btn.dataset.vote;
        btn.disabled = true;
        try {
          const res = await api.post(`/proposals/${proposalId}/vote`, { response: vote });
          toast.success(res?.message || 'Ghi nhận biểu quyết thành công');
          await this.showRoomDetail(this.selectedRoom.id);
        } catch (err) {
          toast.error(err.message);
          btn.disabled = false;
        }
      };
    });
  }

  async handleJoin() {
    if (!authState.isAuthenticated()) {
      toast.info('Vui lòng đăng nhập để tham gia phòng chơi');
      this.onOpenAuthModal();
      return;
    }

    try {
      const res = await api.post(`/rooms/${this.selectedRoom.id}/join`);
      toast.success(res?.message || 'Tham gia phòng thành công!');
      await this.showRoomDetail(this.selectedRoom.id);
    } catch (err) {
      toast.error(err.message);
    }
  }

  async handleLeave() {
    if (!confirm('Bạn có chắc chắn muốn rút khỏi phòng này không?')) return;

    try {
      const res = await api.post(`/rooms/${this.selectedRoom.id}/leave`);
      toast.success(res?.message || 'Thao tác thành công');
      await this.showRoomDetail(this.selectedRoom.id);
    } catch (err) {
      toast.error(err.message);
    }
  }

  async handleCalculatePrice() {
    const promoCode = prompt('Nhập mã khuyến mãi (ví dụ: TUESDAY_30, HSSV_20000 hoặc để trống):', '');
    try {
      const res = await api.post(`/rooms/${this.selectedRoom.id}/calculate-price`, {
        promotion_code: promoCode ? promoCode.trim() : null,
      });
      toast.success('Đã tính toán và chốt hóa đơn snapshot thành công!');
      await this.showRoomDetail(this.selectedRoom.id);
    } catch (err) {
      toast.error(err.message);
    }
  }

        </div>

        <button type="submit" id="btn-submit-create" class="btn btn-primary" style="margin-top: 10px;">
          Tạo phòng ngay
        </button>
      </form>
    `;

    modal.querySelector('#create-room-close').onclick = () => modal.close();

    const form = modal.querySelector('#create-room-form');
    if (preselectedGame) {
      form.querySelector('select[name="game_id"]').value = preselectedGame.id;
    }

    form.onsubmit = async (e) => {
      e.preventDefault();
      const btn = modal.querySelector('#btn-submit-create');
      btn.disabled = true;
      btn.textContent = 'Đang tạo phòng...';

      const formData = new FormData(form);
      const data = Object.fromEntries(formData.entries());

      try {
        const res = await api.post('/rooms', data);
        toast.success('Tạo phòng chơi thành công!');
        modal.close();
        if (res?.data?.room?.id) {
          await this.showRoomDetail(res.data.room.id);
        } else {
          await this.fetchRooms();
        }
      } catch (err) {
        toast.error(err.message);
      } finally {
        btn.disabled = false;
        btn.textContent = 'Tạo phòng ngay';
      }
    };

    document.body.appendChild(modal);
    modal.showModal();
  }

  showProposeModal() {
    const modal = document.createElement('dialog');
    modal.className = 'custom-modal';
    modal.style.cssText = `
      background: var(--color-bg-card, #1e293b);
      color: var(--color-text-main, #f8fafc);
      border: 1px solid var(--color-border, #334155);
      border-radius: 12px;
      padding: 24px;
      width: 100%;
      max-width: 440px;
    `;

    // Gợi ý mốc thời gian 2 ngày sau vào 18h
    const defaultStart = new Date(Date.now() + 2 * 86400000);
    defaultStart.setHours(18, 0, 0, 0);
    const defaultEnd = new Date(defaultStart.getTime() + 4 * 3600000);

    const fmt = (d) => d.toISOString().slice(0, 19).replace('T', ' ');

    modal.innerHTML = `
      <div style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 16px;">
        <h3 style="margin: 0; font-size: 1.2rem;">Đề xuất mốc thời gian chơi</h3>
        <button id="close-prop" style="background: none; border: none; color: #94a3b8; font-size: 1.5rem; cursor: pointer;">&times;</button>
      </div>

      <form id="propose-form" style="display: flex; flex-direction: column; gap: 14px;">
        <div>
          <label style="font-size: 0.85rem; color: #94a3b8; display: block; margin-bottom: 4px;">Thời gian bắt đầu</label>
          <input type="text" name="proposed_start_at" value="${fmt(defaultStart)}" class="form-input" required style="width: 100%; box-sizing: border-box;">
        </div>

        <div>
          <label style="font-size: 0.85rem; color: #94a3b8; display: block; margin-bottom: 4px;">Thời gian kết thúc</label>
          <input type="text" name="proposed_end_at" value="${fmt(defaultEnd)}" class="form-input" required style="width: 100%; box-sizing: border-box;">
        </div>

        <button type="submit" id="btn-submit-prop" class="btn btn-primary" style="margin-top: 10px;">
          Mở biểu quyết mốc giờ
        </button>
      </form>
    `;

    modal.querySelector('#close-prop').onclick = () => modal.close();

    modal.querySelector('#propose-form').onsubmit = async (e) => {
      e.preventDefault();
      const btn = modal.querySelector('#btn-submit-prop');
      btn.disabled = true;
      const data = Object.fromEntries(new FormData(e.target).entries());

      try {
        await api.post(`/rooms/${this.selectedRoom.id}/proposals`, data);
        toast.success('Đã mở biểu quyết mốc giờ chơi mới cho phòng!');
        modal.close();
        await this.showRoomDetail(this.selectedRoom.id);
      } catch (err) {
        toast.error(err.message);
      } finally {
        btn.disabled = false;
      }
    };

    document.body.appendChild(modal);
    modal.showModal();
  }

 * Kịch Bản Sát V16 - Application Security

/**
 * Escape HTML special characters to prevent Cross-Site Scripting (XSS).
 *
 * @param {string|null|undefined} str
 * @returns {string}
 */
export function escapeHtml(str) {
  if (str === null || str === undefined) return '';
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}

/**
 * Validate and sanitize URL to prevent javascript: or data: injection.
 * Only allows http://, https://, or relative / URLs.
 *
 * @param {string|null|undefined} url
 * @returns {string}
 */
export function sanitizeUrl(url) {
  if (!url || typeof url !== 'string') return '';
  const trimmed = url.trim();
  if (/^(https?:\/\/|\/|\.\/)/i.test(trimmed)) {
    return escapeHtml(trimmed);
  }
  return '';
}

/**
 * Lightweight Toast Notification System
 */
class ToastManager {
  constructor() {
    this.container = null;
  }

  ensureContainer() {
    if (!this.container || !document.body.contains(this.container)) {
      this.container = document.createElement('div');
      this.container.id = 'toast-container';
      this.container.style.cssText = `
        position: fixed;
        top: 20px;
        right: 20px;
        z-index: 9999;
        display: flex;
        flex-direction: column;
        gap: 10px;
        max-width: 380px;
        pointer-events: none;
      `;
      document.body.appendChild(this.container);
    }
  }

  show(message, type = 'info', duration = 3500) {
    this.ensureContainer();

    const toast = document.createElement('div');
    toast.className = `toast-item toast-${type}`;
    toast.style.cssText = `
      background: var(--color-bg-card, #1e293b);
      color: var(--color-text-main, #f8fafc);
      padding: 12px 18px;
      border-radius: 8px;
      border: 1px solid var(--color-border, #334155);
      border-left: 5px solid ${type === 'success' ? '#10b981' : type === 'error' ? '#ef4444' : '#3b82f6'};
      box-shadow: 0 10px 15px -3px rgba(0, 0, 0, 0.4);
      font-size: 0.9rem;
      line-height: 1.4;
      pointer-events: auto;
      transition: all 0.3s ease;
      opacity: 0;
      transform: translateY(-10px);
    `;

    toast.innerHTML = `
      <div style="display: flex; align-items: flex-start; justify-content: space-between; gap: 8px;">
        <div>${message}</div>
        <button style="background: none; border: none; color: #94a3b8; cursor: pointer; font-size: 1.1rem; line-height: 1;">&times;</button>
      </div>
    `;

    toast.querySelector('button').onclick = () => this.dismiss(toast);

    this.container.appendChild(toast);

    requestAnimationFrame(() => {
      toast.style.opacity = '1';
      toast.style.transform = 'translateY(0)';
    });

    if (duration > 0) {
      setTimeout(() => this.dismiss(toast), duration);
    }
  }

  dismiss(toast) {
    toast.style.opacity = '0';
    toast.style.transform = 'translateY(-10px)';
    setTimeout(() => {
      if (toast.parentNode) {
        toast.parentNode.removeChild(toast);
      }
    }, 300);
  }

  success(msg) {
    this.show(msg, 'success');
  }

  error(msg) {
    this.show(msg, 'error', 4500);
  }

  info(msg) {
    this.show(msg, 'info');
  }
}

export const toast = new ToastManager();
